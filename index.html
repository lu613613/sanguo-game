联机配置（首次使用）

联机通过你自己的 Supabase 项目同步对局。填入项目地址与 publishable key，验证通过后保存在本机浏览器，之后自动读取。朋友要填同一个项目的地址和 key 才能进同一房间。

第一次使用？先在 Supabase 项目里建一张表（只需一次） 
create table if not exists game_rooms (
  code text primary key,
  state jsonb not null,
  updated_at timestamptz default now()
);
alter table game_rooms enable row level security;
create policy "anyone can read" on game_rooms for select using (true);
create policy "anyone can insert" on game_rooms for insert with check (true);
create policy "anyone can update" on game_rooms for update using (true) with check (true);
create policy "anyone can delete" on game_rooms for delete using (true);
